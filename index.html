import os import time from collections import defaultdict import pandas as pd import requests from IPython.display import FileLink, display # ============================================================ # SETTINGS # ============================================================ LEAGUES = { 2022: "862172111907745792", 2023: "990123475421458432", 2024: "1121308465248542720", 2025: "1251342393802772480", } CURRENT_YEAR = 2025 # Player must be on the same roster for all these weeks. KEEPER_WEEKS = list(range(11, 18)) # Weeks used to retrieve waiver and free-agent transactions. TRANSACTION_WEEKS = range(1, 19) OUTPUT_FILE = "2026_keeper_values.xlsx" BASE_URL = "https://api.sleeper.app/v1" TIMEOUT = 60 session = requests.Session() api_cache = {} # ============================================================ # GENERAL HELPERS # ============================================================ def log(message=""): print(message, flush=True) def get_json(url, description): """Download JSON with progress, retries, and caching.""" if url in api_cache: return api_cache[url] log(f"Loading: {description}") last_error = None for attempt in range(1, 4): try: start = time.time() response = session.get( url, timeout=TIMEOUT, ) response.raise_for_status() data = response.json() api_cache[url] = data count = ( len(data) if isinstance(data, (list, dict)) else 0 ) log( f"Completed: {description} — " f"{count:,} records — " f"{time.time() - start:.1f}s" ) return data except ( requests.RequestException, ValueError, ) as error: last_error = error if attempt < 3: log( f"Attempt {attempt} failed. Retrying..." ) time.sleep(attempt * 2) raise RuntimeError( f"Could not load {description}\n" f"URL: {url}\n" f"Error: {last_error}" ) def league_url(league_id, endpoint): return f"{BASE_URL}/league/{league_id}/{endpoint}" def to_float(value, default=0.0): try: return float(value) except (TypeError, ValueError): return default def safe_width(series, column_name): """Calculate Excel column width safely.""" lengths = series.apply( lambda value: ( len(str(value)) if pd.notna(value) else 0 ) ) data_width = ( int(lengths.max()) if not lengths.empty else 0 ) return min( max(data_width, len(str(column_name))) + 2, 40, ) # ============================================================ # AUCTION VALUE # ============================================================ def extract_auction(metadata): """ Extract the auction price from Sleeper draft metadata. """ if not isinstance(metadata, dict): return 0.0 preferred_terms = ( "amount", "auction", "bid", "price", "cost", ) # Check likely auction fields first. for key, value in metadata.items(): key_text = str(key).lower() if any( term in key_text for term in preferred_terms ): number = to_float(value, None) if number is not None: return number # Preserve the fallback behavior from the original code. for value in metadata.values(): number = to_float(value, None) if number is not None: return number return 0.0 # ============================================================ # STEP 1: LOAD NFL PLAYER DATABASE # ============================================================ log("=" * 70) log("STEP 1: Loading NFL player database") log("=" * 70) players = get_json( f"{BASE_URL}/players/nfl", "NFL player database", ) def player_name(player_id): player_id = str(player_id) info = players.get(player_id, {}) if info.get("full_name"): return info["full_name"] first_name = info.get("first_name", "") last_name = info.get("last_name", "") combined = f"{first_name} {last_name}".strip() return combined or player_id def player_team(player_id): return players.get( str(player_id), {}, ).get("team", "") def player_position(player_id): return players.get( str(player_id), {}, ).get("position", "") def games_played(player_id): """ Sleeper's player database may not contain season game totals. Missing values default to 1, matching the original code. """ value = players.get( str(player_id), {}, ).get("games_played", 1) try: return int(value or 1) except (TypeError, ValueError): return 1 # ============================================================ # STEP 2: LOAD USERS AND ROSTERS # ============================================================ log("\n" + "=" * 70) log("STEP 2: Loading users and rosters") log("=" * 70) users_by_year = {} rosters_by_year = {} roster_owner_by_year = {} for year, league_id in LEAGUES.items(): log(f"\nProcessing {year}") users = get_json( league_url(league_id, "users"), f"{year} users", ) users_by_year[year] = { str(user["user_id"]): user.get( "display_name", "Unknown", ) for user in users } rosters = get_json( league_url(league_id, "rosters"), f"{year} rosters", ) rosters_by_year[year] = rosters roster_owner_by_year[year] = { roster["roster_id"]: str(roster["owner_id"]) for roster in rosters if roster.get("owner_id") is not None } # ============================================================ # STEP 3: LOAD DRAFTS # ============================================================ log("\n" + "=" * 70) log("STEP 3: Loading drafts") log("=" * 70) draft_picks_by_year = {} selected_drafts = {} def get_best_draft(year, league_id): """ Select the draft with the most auction-priced picks. """ drafts = get_json( league_url(league_id, "drafts"), f"{year} draft list", ) if not drafts: log(f"WARNING: No draft found for {year}") return None, [] candidates = [] for number, draft in enumerate(drafts, start=1): draft_id = draft.get("draft_id") if not draft_id: continue picks = get_json( f"{BASE_URL}/draft/{draft_id}/picks", f"{year} draft candidate {number}", ) priced_picks = sum( extract_auction( pick.get("metadata") or {} ) > 0 for pick in picks ) completed = int( str( draft.get("status", "") ).lower() == "complete" ) candidates.append({ "draft": draft, "picks": picks, "priced_picks": priced_picks, "completed": completed, }) log( f"{year} candidate {number}: " f"{len(picks):,} picks, " f"{priced_picks:,} priced picks, " f"status={draft.get('status', 'unknown')}" ) if not candidates: return None, [] best = max( candidates, key=lambda item: ( item["priced_picks"], len(item["picks"]), item["completed"], ), ) log( f"Selected {year} draft: " f"{best['draft'].get('draft_id')}" ) return best["draft"], best["picks"] for year, league_id in LEAGUES.items(): draft, picks = get_best_draft( year, league_id, ) selected_drafts[year] = draft draft_picks_by_year[year] = picks # ============================================================ # STEP 4: BUILD DRAFT HISTORY # ============================================================ log("\n" + "=" * 70) log("STEP 4: Building draft history") log("=" * 70) draft_rows = [] for year, picks in draft_picks_by_year.items(): users = users_by_year[year] roster_owners = roster_owner_by_year[year] for pick_number, pick in enumerate( picks, start=1, ): player_id = str( pick.get("player_id", "") ) owner_id = pick.get("picked_by") if owner_id is not None: owner_id = str(owner_id) if not owner_id: owner_id = roster_owners.get( pick.get("roster_id") ) draft_rows.append({ "Owner": users.get( owner_id, "Unknown", ), "Owner_ID": owner_id, "Year": year, "Round": pick.get("round", ""), "Team": player_team(player_id), "Position": player_position(player_id), "Player": player_name(player_id), "Player_ID": player_id, "Cost": extract_auction( pick.get("metadata") or {} ), }) if ( pick_number % 25 == 0 or pick_number == len(picks) ): log( f"{year}: processed " f"{pick_number:,} of " f"{len(picks):,} picks" ) df_drafts = pd.DataFrame(draft_rows) if df_drafts.empty: raise RuntimeError( "No draft records were returned." ) df_drafts = ( df_drafts[ [ "Owner", "Owner_ID", "Year", "Round", "Team", "Position", "Player", "Player_ID", "Cost", ] ] .sort_values( ["Year", "Owner", "Round", "Cost"], ascending=[False, True, True, False], ) .reset_index(drop=True) ) log( f"Draft history completed: " f"{len(df_drafts):,} records" ) # ============================================================ # STEP 5: DETERMINE KEEPER ELIGIBILITY # ============================================================ log("\n" + "=" * 70) log("STEP 5: Determining keeper eligibility") log("=" * 70) def get_keeper_eligible(year, league_id): """ A player must appear on the same roster during every week from Week 11 through Week 17. """ roster_history = defaultdict( lambda: defaultdict(set) ) for position, week in enumerate( KEEPER_WEEKS, start=1, ): matchups = get_json( league_url( league_id, f"matchups/{week}", ), f"{year} Week {week} matchups", ) for entry in matchups: roster_id = entry.get("roster_id") for player_id in entry.get("players") or []: roster_history[roster_id][ str(player_id) ].add(week) log( f"{year}: completed keeper week " f"{position} of {len(KEEPER_WEEKS)}" ) required_weeks = set(KEEPER_WEEKS) records = [] for roster_id, roster_players in roster_history.items(): owner_id = roster_owner_by_year[year].get( roster_id ) if owner_id is None: continue manager = users_by_year[year].get( owner_id, "Unknown", ) for player_id, weeks_present in roster_players.items(): if required_weeks.issubset(weeks_present): records.append({ "Year": year, "Owner_ID": owner_id, "Manager": manager, "Roster_ID": roster_id, "Player_ID": player_id, }) result = pd.DataFrame( records, columns=[ "Year", "Owner_ID", "Manager", "Roster_ID", "Player_ID", ], ) log( f"{year}: found " f"{len(result):,} keeper-eligible players" ) return result eligible_by_year = { year: get_keeper_eligible( year, league_id, ) for year, league_id in LEAGUES.items() } # ============================================================ # STEP 6: DETERMINE KEEPER YEAR # ============================================================ log("\n" + "=" * 70) log("STEP 6: Determining keeper years") log("=" * 70) keepers_2026 = eligible_by_year[CURRENT_YEAR].copy() if keepers_2026.empty: raise RuntimeError( "No 2026-eligible players were found." ) def was_eligible(year, owner_id, player_id): prior = eligible_by_year.get(year) if prior is None or prior.empty: return False return ( (prior["Owner_ID"] == owner_id) & (prior["Player_ID"] == player_id) ).any() def prior_keeper_years(owner_id, player_id): """ Keeper use in 2026: 0 prior consecutive years = Keeper Year 1 1 prior consecutive year = Keeper Year 2 2 prior consecutive years = Keeper Year 3 3 prior consecutive years = Keeper Year 4, not permitted """ if not was_eligible( 2024, owner_id, player_id, ): return 0 if not was_eligible( 2023, owner_id, player_id, ): return 1 if not was_eligible( 2022, owner_id, player_id, ): return 2 return 3 keepers_2026["Prior_Keeper_Years"] = ( keepers_2026.apply( lambda row: prior_keeper_years( row["Owner_ID"], row["Player_ID"], ), axis=1, ) ) # A fourth keeper use is not permitted. excluded_fourth_year = keepers_2026[ keepers_2026["Prior_Keeper_Years"] >= 3 ].copy() # Keeper Years 1, 2, and 3 remain eligible. keepers_2026 = keepers_2026[ keepers_2026["Prior_Keeper_Years"] < 3 ].copy() keepers_2026["Keeper_Year"] = ( keepers_2026["Prior_Keeper_Years"] + 1 ) log( f"First-, second-, or third-year keepers: " f"{len(keepers_2026):,}" ) log( f"Excluded fourth-year players: " f"{len(excluded_fourth_year):,}" ) # ============================================================ # STEP 7: MAP 2025 AUCTION VALUES # ============================================================ log("\n" + "=" * 70) log("STEP 7: Mapping 2025 auction values") log("=" * 70) auction_values = {} for pick in draft_picks_by_year[CURRENT_YEAR]: player_id = str( pick.get("player_id", "") ) price = extract_auction( pick.get("metadata") or {} ) auction_values[player_id] = max( price, auction_values.get(player_id, 0), ) keepers_2026["Auction_Value"] = ( keepers_2026["Player_ID"] .map(auction_values) .fillna(0.0) ) log( f"Players with positive auction values: " f"{(keepers_2026['Auction_Value'] > 0).sum():,}" ) # ============================================================ # STEP 8: CALCULATE 2025 WAIVER VALUES # ============================================================ log("\n" + "=" * 70) log("STEP 8: Calculating 2025 waiver values") log("=" * 70) league_2025 = LEAGUES[CURRENT_YEAR] waiver_keeper_costs = {} for week in TRANSACTION_WEEKS: transactions = get_json( league_url( league_2025, f"transactions/{week}", ), f"2025 Week {week} transactions", ) additions = 0 for transaction in transactions: if transaction.get("type") not in ( "waiver", "free_agent", ): continue adds = transaction.get("adds") or {} if not adds: continue settings = transaction.get("settings") or {} waiver_bid = to_float( settings.get("waiver_bid", 0), 0, ) # League rule: # # Waiver base = max(waiver bid × 2, $10) # Keeper cost = waiver base + $5 # # $0 bid -> $10 base + $5 = $15 # $3 bid -> $10 base + $5 = $15 # $5 bid -> $10 base + $5 = $15 # $8 bid -> $16 base + $5 = $21 waiver_base = max( waiver_bid * 2, 10, ) transaction_keeper_cost = ( waiver_base + 5 ) for player_id in adds: player_id = str(player_id) # Preserve the highest waiver-derived value # if the player was added multiple times. waiver_keeper_costs[player_id] = max( transaction_keeper_cost, waiver_keeper_costs.get( player_id, 0, ), ) additions += 1 log( f"Week {week}: " f"{additions:,} additions processed" ) # ============================================================ # STEP 9: CALCULATE FINAL 2026 KEEPER COST # ============================================================ log("\n" + "=" * 70) log("STEP 9: Calculating 2026 keeper costs") log("=" * 70) def calculate_keeper_cost(row): """ Drafted player: max(draft price, $5) + $5 Waiver/free-agent player: max(waiver bid × 2, $10) + $5 If both values exist: use the higher value. Keeper Years 1, 2, and 3 use the same $5 keeper fee. """ player_id = row["Player_ID"] auction_value = to_float( row["Auction_Value"], 0, ) draft_keeper_cost = None if auction_value > 0: draft_base = max( auction_value, 5, ) draft_keeper_cost = ( draft_base + 5 ) waiver_keeper_cost = ( waiver_keeper_costs.get(player_id) ) possible_costs = [ value for value in ( draft_keeper_cost, waiver_keeper_cost, ) if value is not None ] # If no draft or waiver record is found: # $10 minimum base plus $5 keeper fee. final_cost = ( max(possible_costs) if possible_costs else 15 ) if ( draft_keeper_cost is not None and waiver_keeper_cost is not None ): cost_source = ( "Draft" if draft_keeper_cost >= waiver_keeper_cost else "Waiver" ) elif draft_keeper_cost is not None: cost_source = "Draft" elif waiver_keeper_cost is not None: cost_source = "Waiver" else: cost_source = "Minimum" return pd.Series({ "Draft_Keep_Cost": draft_keeper_cost, "Waiver_Keep_Cost": waiver_keeper_cost, "2026_Keep_Cost": final_cost, "Cost_Source": cost_source, }) cost_details = keepers_2026.apply( calculate_keeper_cost, axis=1, ) keepers_2026 = pd.concat( [ keepers_2026.reset_index(drop=True), cost_details.reset_index(drop=True), ], axis=1, ) # ============================================================ # STEP 10: ADD PLAYER DETAILS # ============================================================ log("\n" + "=" * 70) log("STEP 10: Adding player details") log("=" * 70) keepers_2026["Player_Name"] = ( keepers_2026["Player_ID"] .map(player_name) ) keepers_2026["Team"] = ( keepers_2026["Player_ID"] .map(player_team) ) keepers_2026["Position"] = ( keepers_2026["Player_ID"] .map(player_position) ) keepers_2026["Games_Played"] = ( keepers_2026["Player_ID"] .map(games_played) ) keepers_2026 = ( keepers_2026[ [ "Manager", "Player_Name", "Team", "Position", "Player_ID", "Keeper_Year", "Auction_Value", "Draft_Keep_Cost", "Waiver_Keep_Cost", "2026_Keep_Cost", "Cost_Source", "Games_Played", ] ] .drop_duplicates( subset=[ "Manager", "Player_ID", ] ) .sort_values( [ "Manager", "Position", "Player_Name", ] ) .reset_index(drop=True) ) # ============================================================ # STEP 11: FORMAT EXCLUDED FOURTH-YEAR PLAYERS # ============================================================ if not excluded_fourth_year.empty: excluded_fourth_year["Player_Name"] = ( excluded_fourth_year["Player_ID"] .map(player_name) ) excluded_fourth_year["Team"] = ( excluded_fourth_year["Player_ID"] .map(player_team) ) excluded_fourth_year["Position"] = ( excluded_fourth_year["Player_ID"] .map(player_position) ) excluded_fourth_year = ( excluded_fourth_year[ [ "Manager", "Player_Name", "Team", "Position", "Player_ID", "Prior_Keeper_Years", ] ] .sort_values( [ "Manager", "Player_Name", ] ) .reset_index(drop=True) ) # ============================================================ # STEP 12: PREVIEW AND VALIDATION # ============================================================ log("\n" + "=" * 70) log("STEP 12: Previewing results") log("=" * 70) log( f"Eligible keeper records: " f"{len(keepers_2026):,}" ) log( f"Managers represented: " f"{keepers_2026['Manager'].nunique():,}" ) log( f"Players with draft values: " f"{(keepers_2026['Auction_Value'] > 0).sum():,}" ) log( f"Players with waiver values: " f"{keepers_2026['Waiver_Keep_Cost'].notna().sum():,}" ) log("\n2026 keeper preview:") display( keepers_2026.head(40) ) log("\nEligible players per manager:") display( keepers_2026 .groupby("Manager") .size() .reset_index( name="Eligible_Players" ) ) # Useful check for Caleb Williams. caleb_check = keepers_2026[ keepers_2026["Player_Name"] .str.contains( "Caleb Williams", case=False, na=False, ) ] if not caleb_check.empty: log("\nCaleb Williams check:") display(caleb_check) # ============================================================ # STEP 13: EXPORT TO EXCEL # ============================================================ log("\n" + "=" * 70) log("STEP 13: Exporting Excel workbook") log("=" * 70) with pd.ExcelWriter( OUTPUT_FILE, engine="xlsxwriter", ) as writer: keepers_2026.to_excel( writer, sheet_name="2026 Eligible Keepers", index=False, ) df_drafts.to_excel( writer, sheet_name="Draft History", index=False, ) if not excluded_fourth_year.empty: excluded_fourth_year.to_excel( writer, sheet_name="Excluded Fourth Year", index=False, ) workbook = writer.book header_format = workbook.add_format({ "bold": True, "border": 1, "align": "center", }) money_format = workbook.add_format({ "num_format": "$0.00", }) integer_format = workbook.add_format({ "num_format": "0", }) sheets = { "2026 Eligible Keepers": keepers_2026, "Draft History": df_drafts, } if not excluded_fourth_year.empty: sheets["Excluded Fourth Year"] = ( excluded_fourth_year ) for sheet_name, dataframe in sheets.items(): worksheet = writer.sheets[sheet_name] for column_number, column_name in enumerate( dataframe.columns ): worksheet.write( 0, column_number, column_name, header_format, ) worksheet.set_column( column_number, column_number, safe_width( dataframe[column_name], column_name, ), ) worksheet.freeze_panes(1, 0) if not dataframe.empty: worksheet.autofilter( 0, 0, len(dataframe), len(dataframe.columns) - 1, ) keeper_sheet = writer.sheets[ "2026 Eligible Keepers" ] # Keeper Year keeper_sheet.set_column( "F:F", 12, integer_format, ) # Auction and keeper cost columns keeper_sheet.set_column( "G:J", 18, money_format, ) # Draft History Cost column writer.sheets["Draft History"].set_column( "I:I", 12, money_format, ) # ============================================================ # COMPLETE # ============================================================ log("\n" + "=" * 70) log("COMPLETE") log("=" * 70) log( f"Saved workbook:\n" f"{os.path.abspath(OUTPUT_FILE)}" ) log( f"Eligible 2026 keepers: " f"{len(keepers_2026):,}" ) log( f"Excluded fourth-year players: " f"{len(excluded_fourth_year):,}" ) display( FileLink(OUTPUT_FILE) )
